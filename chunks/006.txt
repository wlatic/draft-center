udes(pos) &&
       !before.some(x=>String(x?.metadata?.position||"").toUpperCase()===pos) &&
       storyOnce(`first:${pos}`)){
      return {kicker:"DRAFT MILESTONE",title:`FIRST ${pos} OFF THE BOARD`,sub:`${pickName(p)} starts the ${pos} market.`,tone:"gold",duration:3500};
    }

    const prev=liveHistory[liveHistory.length-2];
    if(prev && String(prev.picked_by||"")===String(p.picked_by||"")){
      const prevPos=String(prev?.metadata?.position||"").toUpperCase();
      if(prevPos===pos && storyAllowed(`double:${owner.uid||owner.display_name}:${pos}`,Number(p.pick_no),8)){
        return {kicker:"DOUBLE DIP",title:`BACK-TO-BACK ${pos}s`,sub:`${broadcastName(owner)} uses the turn to take two ${pos}s.`,tone:"cyan",duration:3500};
      }
    }

    const nfl=String(p?.metadata?.team||"");
    if(owner.uid && nfl){
      const own=picks.filter(x=>String(x.picked_by||"")===String(owner.uid));
      const same=own.filter(x=>String(x?.metadata?.team||"")===nfl);
      const qb=same.find(x=>String(x?.metadata?.position||"").toUpperCase()==="QB");
      const pass=same.find(x=>["WR","TE"].includes(String(x?.metadata?.position||"").toUpperCase()));
      if(qb&&pass&&(qb===p||pass===p) && storyOnce(`stack:${owner.uid}:${nfl}`)){
        return {kicker:"STACK ALERT",title:`${pickName(qb)} + ${pickName(pass)}`,sub:`${broadcastName(owner)} has a ${nfl} QB/pass-catcher stack.`,tone:"cyan",duration:3600};
      }
      if((same.length===3 || same.length===5) && storyOnce(`homer:${owner.uid}:${nfl}:${same.length}`)){
        return {kicker:"HOMER ALERT",title:`${same.length} ${nfl} PLAYERS`,sub:`${broadcastName(owner)} is building the ${nfl}.`,tone:"gold",duration:3400};
      }
    }

    const c=rosterCounts(owner.uid);
    const currentRound=info(Number(p.pick_no)).round;
    if(currentRound>=5 && c.WR>=4 && c.RB<=1 && storyOnce(`identity:zeroRB:${owner.uid}`)){
      return {kicker:"ROSTER IDENTITY",title:"ZERO-RB WATCH",sub:`${broadcastName(owner)} now has ${c.WR} WRs and ${c.RB} RB.`,tone:"cyan",duration:3300};
    }
    if(currentRound>=5 && c.RB>=4 && c.WR<=1 && storyOnce(`identity:ground:${owner.uid}`)){
      return {kicker:"ROSTER IDENTITY",title:"GROUND GAME",sub:`${broadcastName(owner)} now has ${c.RB} RBs and ${c.WR} WR.`,tone:"green",duration:3300};
    }

    return null;
  }

  function roundSummary(round){
    const teams=Number(draft?.settings?.teams||10);
    const arr=picks.slice((round-1)*teams,round*teams);
    const c={};
    for(const p of arr){const pos=String(p?.metadata?.position||"OTHER").toUpperCase();c[pos]=(c[pos]||0)+1;}
    return ["QB","RB","WR","TE","K","DEF","OTHER"].filter(k=>c[k]).map(k=>`${c[k]} ${k}`).join(" · ");
  }



  function roundLeader(round){
    const teams=Number(draft?.settings?.teams||10);
    const arr=picks.slice((round-1)*teams,round*teams);
    const counts={};
    for(const p of arr){
      const pos=String(p?.metadata?.position||"OTHER").toUpperCase();
      counts[pos]=(counts[pos]||0)+1;
    }
    const entries=Object.entries(counts).sort((a,b)=>b[1]-a[1]);
    return entries[0]||null;
  }

  function roundBestValue(round){
    if(!marketReady)return null;
    const teams=Number(draft?.settings?.teams||10);
    const arr=picks.slice((round-1)*teams,round*teams);
    let best=null;
    for(const p of arr){
      const mv=marketValue(p);
      if(!mv)continue;
      const delta=Number(p.pick_no)-mv.rank;
      if(!best || delta>best.delta)best={p,delta,adp:mv.rank};
    }
    return best && best.delta>=8 ? best : null;
  }

  function bestAvailable(){
    if(!marketReady || !adpRows.length)return null;
    const taken=new Set(picks.map(p=>String(p.player_id)));
    for(const x of adpRows){
      if(taken.has(String(x.player_id)))continue;
      if(!["QB","RB","WR","TE","K","DEF"].some(pos=>(x.fantasy_positions||[]).includes(pos)))continue;
      return x;
    }
    return null;
  }

  function slipStory(){
    const x=bestAvailable();
    if(!x)return null;
    const nextPick=liveProgress().ordinal;
    const slip=nextPick-x.adp;

    // Treat "still available" as a studio note, not something to repeat every pick.
    if(slip<15 || alertedSlips.has(String(x.player_id)) || nextPick-lastSlipStoryPick<8)return null;
    alertedSlips.add(String(x.player_id));
    lastSlipStoryPick=nextPick;

    return {
      kicker:"STILL ON THE BOARD",
      title:x.full_name||"Top ADP player",
      sub:`Sleeper ADP ${x.adp.toFixed(1)} · draft has reached pick ${nextPick}.`,
      image:x.player_id?`https://sleepercdn.com/content/nfl/players/${encodeURIComponent(x.player_id)}.jpg`:"",
      fallback:initials(x.full_name),tone:"cyan",duration:3500,
      stats:[["Sleeper ADP",x.adp.toFixed(1)],["Next pick",nextPick],["Slide",Math.round(slip)+" picks"]]
    };
  }

  function quickPickStory(p,seconds){
    if(!Number.isFinite(seconds))return null;

    // Sleeper's documented pick payload has no guaranteed per-pick "autodraft"
    // flag. If CPU autopick is enabled globally, do not run speed stories at all.
    if(Number(draft?.settings?.cpu_autopick||0)===1)return null;
    if(isAutodraftManager(ownerForPick(p)))return null;

    // Honor any explicit auto marker if Sleeper ever includes one.
    const explicitAuto =
      p?.is_auto===true || p?.is_autopick===true ||
      String(p?.metadata?.is_auto||"")==="1" ||
      String(p?.metadata?.is_autopick||"")==="1";
    if(explicitAuto)return null;

    // Near-instant choices are too ambiguous (force CPU, queue/automation, etc.).
    // Timer-expiry choices also aren't "quick picks".
    if(seconds<2.5 || seconds>8)return null;

    const round=info(Number(p.pick_no)).round;
    if(!storyOnce(`quick:round:${round}`))return null;

    return {
      kicker:"QUICK TURNAROUND",
      title:`${seconds.toFixed(1)} SECONDS`,
      sub:`${pickName(p)} came off the board almost immediately.`,
      tone:"cyan",duration:2600
    };
  }


  function rememberDeferred(story){
    if(!story)return;

    // Round recaps are allowed to linger longer than normal analysis.
    const ttl=story.roundRecap?90000:45000;
    const key=story.storyKey || `${story.kicker}:${story.title}`;

    // Deduplicate, preferring the newest version of the same story.
    for(let i=deferredStories.length-1;i>=0;i--){
      if((deferredStories[i].storyKey || `${deferredStories[i].kicker}:${deferredStories[i].title}`)===key){
        deferredStories.splice(i,1);
      }
    }

    deferredStories.push({
      ...story,
      storyKey:key,
      deferredAt:Date.now(),
      deferredExpiresAt:Date.now()+ttl,
      replayCount:Number(story.replayCount||0)
    });

    deferredStories.sort((a,b)=>(b.priority||0)-(a.priority||0) || b.deferredAt-a.deferredAt);
    while(deferredStories.length>DEFERRED_MAX)deferredStories.pop();
  }
