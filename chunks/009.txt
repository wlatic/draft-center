   });
    }

    return facts.sort((a,b)=>b.score-a.score).slice(0,4);
  }

  function roundFacts(){
    const round=currentRoundNumber();
    const rp=picksForRound(round);
    const c=positionBreakdown(rp);
    const facts=[];

    const byCount=Object.entries(c)
      .filter(([,n])=>n>0)
      .sort((a,b)=>b[1]-a[1]);

    for(const [pos,count] of byCount.slice(0,3)){
      facts.push({
        score:count,
        title:`${count} ${pos}${count===1?"":"s"} this round`,
        detail:`${rp.length} of ${Number(draft?.settings?.teams||10)} picks complete`,
        tone:count>=4?"hot":""
      });
    }

    if(rp.length){
      const first=rp[0];
      facts.push({
        score:.1,
        title:`Round ${round} opened with ${pickName(first)}`,
        detail:`${first?.metadata?.position||"—"} · ${first?.metadata?.team||"FA"}`,
        tone:""
      });
    }

    if(!facts.length){
      facts.push({
        score:0,
        title:`Round ${round} just started`,
        detail:"Waiting for the board to develop",
        tone:""
      });
    }

    return facts.slice(0,4);
  }

  function keeperFacts(){
    const keepers=picks.filter(isKeeperPick);
    if(!keepers.length)return [];
    const facts=[];

    const pos=positionBreakdown(keepers);
    const byCount=Object.entries(pos).filter(([,n])=>n>0).sort((a,b)=>b[1]-a[1]);
    for(const [p,n] of byCount.slice(0,3)){
      facts.push({
        score:n,
        title:`${n} keeper ${p}${n===1?"":"s"}`,
        detail:"Pre-filled before the live draft",
        tone:"good"
      });
    }

    facts.push({
      score:.1,
      title:`${keepers.length} keeper slots total`,
      detail:"They do not advance the live pick sequence",
      tone:""
    });

    return facts.slice(0,4);
  }

  function buildInsightPanels(){
    const panels=[];

    const round=roundFacts();
    if(round.length) panels.push({label:"This Round",kicker:`ROUND ${currentRoundNumber()}`,rows:round});

    const market=marketMoverFacts();
    if(market.length) panels.push({label:"Market Movers",kicker:"ADP DESK",rows:market});

    const positions=positionWatchFacts();
    if(positions.length) panels.push({label:"Position Watch",kicker:"DRAFT TRENDS",rows:positions});

    const rosters=currentRosterFacts();
    if(rosters.length) panels.push({label:"Team Trends",kicker:"ROSTER BUILDS",rows:rosters});

    const stacks=stackFacts();
    if(stacks.length) panels.push({label:"Connections",kicker:"STACKS & HOMERS",rows:stacks});

    const keeper=keeperFacts();
    if(keeper.length) panels.push({label:"Keeper Notes",kicker:"KEEPER BOARD",rows:keeper});

    return panels;
  }

  function renderInsight(){
    if(!draft)return;

    const panels=buildInsightPanels();
    insightSlides=panels;

    if(!panels.length)return;
    if(insightIndex>=panels.length)insightIndex=0;

    const panel=panels[insightIndex];
    $("insightLabel").textContent=panel.label||"DraftCenter Desk";
    $("insightKicker").textContent=panel.kicker||"LIVE";

    $("insightList").innerHTML=(panel.rows||[]).slice(0,4).map((row,i)=>`
      <div class="insight-row ${esc(row.tone||"")}">
        <div class="insight-rank">${i+1}</div>
        <div class="insight-row-copy">
          <div class="insight-row-title">${esc(row.title||"")}</div>
          <div class="insight-row-detail">${esc(row.detail||"")}</div>
        </div>
      </div>
    `).join("");

    $("insightDots").innerHTML=panels
      .map((_,i)=>`<span class="insight-dot${i===insightIndex?" active":""}"></span>`)
      .join("");
  }

  function rotateInsight(){
    if(!insightSlides.length){
      renderInsight();
      return;
    }
    insightIndex=(insightIndex+1)%insightSlides.length;
    renderInsight();
  }

  function soundsWithTags(tags){
    const wanted=Array.isArray(tags)?tags:[tags];
    return SOUND_CATALOG.filter(s=>wanted.every(t=>s.tags.includes(t)));
  }

  function chooseSound(tags){
    const matches=soundsWithTags(tags);
    if(!matches.length)return null;
    let choices=matches.filter(s=>!soundRecentIds.includes(s.id));
    if(!choices.length)choices=matches;
    const chosen=choices[Math.floor(Math.random()*choices.length)];
    soundRecentIds.push(chosen.id);
    while(soundRecentIds.length>4)soundRecentIds.shift();
    return chosen;
  }

  function makeRepoAudio(sound,volume=1,rate=1){
    if(!sound)return null;
    const a=new Audio(sound.url);
    a.preload="auto";
    a.volume=Math.max(0,Math.min(1,volume*masterVolume));
    a.playbackRate=Math.max(.85,Math.min(1.2,rate));
    return a;
  }

  function cleanupFinishedClips(){
    activeClipAudios=activeClipAudios.filter(a=>!a.ended&&!a.paused);
  }

  async function playTags(tags,volume=1,rate=1){
    if(!soundEnabled)return null;
    cleanupFinishedClips();
    const chosen=chooseSound(tags);
    if(!chosen)return null;
    const a=makeRepoAudio(chosen,volume,rate);
    activeClipAudios.push(a);
    try{await a.play()}catch(_){}
    return a;
  }

  function updateAudioState(){
    const el=$("audioState");
    if(el)el.value=soundEnabled?"ready — CC0 repo":"off";
  }
