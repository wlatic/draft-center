    n=>`${n} is absent. The chair is empty; the picks are still legally binding.`,
    n=>`${n} couldn't be here, but their roster has bravely continued without adult supervision.`,
    n=>`${n} is drafting from afar. Very brave. Very convenient.`
  ];

  const SPICY_ROASTS = [
    n=>`${n} couldn't be bothered to attend their own draft. Commitment remains undefeated.`,
    n=>`${n} is absent, which at least gives them a ready-made excuse for this roster.`,
    n=>`${n}'s chair is empty and somehow still showing more draft awareness than expected.`,
    n=>`${n} couldn't make the draft. Their strategy apparently couldn't make it either.`,
    n=>`${n} is drafting remotely — because facing the room after these picks would be awkward.`,
    n=>`${n} has outsourced both attendance and accountability tonight.`,
    n=>`${n} isn't here. Conveniently, neither is anyone who can explain the plan.`,
    n=>`${n} skipped draft night. Bold choice for someone who'll talk about injuries for four months.`
  ];

  function absentRoastStory(m,pickNo){
    if(roastLevel==="off" || !isAbsentManager(m))return null;

    const uid=m.uid||managerUsername(m)||managerTitle(m);
    const last=Number(roastMemory.get(uid)??-999);

    // At most once every ~2 rounds / 18 picks per absent manager.
    if(Number(pickNo)-last<18)return null;
    roastMemory.set(uid,Number(pickNo));

    const arr=roastLevel==="light"?LIGHT_ROASTS:SPICY_ROASTS;
    const index=Math.abs(Number(pickNo)+String(uid).length)%arr.length;
    const copy=arr[index](roastName(m));

    return {
      kicker:"ABSENT OWNER DESK",
      title:String(roastName(m)).toUpperCase(),
      sub:copy,
      tone:"gold",
      duration:3900,
      priority:52,
      storyKey:`absent:${uid}:${Math.floor(Number(pickNo)/18)}`
    };
  }

  function managerTitle(m){ return m?.team_name||m?.display_name||"Unknown"; }
  function managerSub(m){ return m?.team_name && m?.display_name && m.team_name!==m.display_name ? m.display_name : ""; }

  function setImgBox(el, src, fallback){
    el.replaceChildren();
    if(src){
      const img=document.createElement("img");
      img.alt="";
      img.src=src;
      img.onerror=()=>{el.replaceChildren(document.createTextNode(fallback));};
      el.appendChild(img);
    }else el.textContent=fallback;
  }

  function rosterCounts(uid){
    const c={QB:0,RB:0,WR:0,TE:0,K:0,DEF:0};
    if(!uid) return c;
    for(const p of picks){
      if(String(p.picked_by||"")!==String(uid)) continue;
      const pos=String(p?.metadata?.position||"").toUpperCase();
      if(c[pos]!==undefined)c[pos]++;
    }
    return c;
  }

  function rosterLine(uid){
    const c=rosterCounts(uid);
    const x=["QB","RB","WR","TE"].map(k=>`${k} ${c[k]}`);
    if(c.K)x.push(`K ${c.K}`);
    if(c.DEF)x.push(`DEF ${c.DEF}`);
    return x.join("  ·  ");
  }

  function posClass(pos){
    const p=String(pos||"").toLowerCase();
    return ["qb","rb","wr","te","k","def"].includes(p)?p:"other";
  }


  function storyAllowed(key,pickNo,gap){
    const last=Number(storyMemory.get(key)??-999);
    if(Number(pickNo)-last<gap)return false;
    storyMemory.set(key,Number(pickNo));
    return true;
  }

  function storyOnce(key){
    if(storyMemory.has(key))return false;
    storyMemory.set(key,picks.length);
    return true;
  }

  function marketStory(p){
    const mv=marketValue(p);
    if(!mv)return null;

    const cell=cellForPick(p);
    const pickNo=Number(cell?.ordinal || p?.pick_no || 0);
    const delta=pickNo-mv.rank;
    const abs=Math.abs(delta);

    // Big discrepancies can always break through. Smaller ones get a cooldown
    // so DraftCenter doesn't become "VALUE ALERT" every other pick.
    if(abs<12)return null;
    if(abs<25 && !storyAllowed(delta>0?"market:value":"market:early",pickNo,4)) return null;

    if(delta>=25){
      return {kicker:"DRAFTCENTER VALUE",title:`${pickName(p)} FALLS ${Math.round(delta)} PICKS`,sub:`Selected ${pickNo} · Sleeper ADP ${mv.rank.toFixed(1)} · ${mv.source}`,tone:"green",duration:3900,
        stats:[["Selected",pickNo],["Sleeper ADP",mv.rank.toFixed(1)],["Value","+ "+Math.round(delta)+" picks"]]};
    }
    if(delta>=12){
      return {kicker:"VALUE ALERT",title:`${Math.round(delta)} PICKS PAST ADP`,sub:`${pickName(p)} stayed on the board longer than Sleeper's market expected.`,tone:"green",duration:3500,
        stats:[["Selected",pickNo],["Sleeper ADP",mv.rank.toFixed(1)]]};
    }
    if(delta<=-25){
      return {kicker:"AGGRESSIVE PICK",title:`${Math.abs(Math.round(delta))} PICKS AHEAD OF ADP`,sub:`${pickName(p)} goes well ahead of the Sleeper market.`,tone:"red",duration:3900,
        stats:[["Selected",pickNo],["Sleeper ADP",mv.rank.toFixed(1)]]};
    }
    if(delta<=-12){
      return {kicker:"EARLY CALL",title:`${pickName(p)} GOES EARLY`,sub:`Selected ${pickNo} · Sleeper ADP ${mv.rank.toFixed(1)}.`,tone:"gold",duration:3500,
        stats:[["Selected",pickNo],["Sleeper ADP",mv.rank.toFixed(1)],["Difference",Math.abs(Math.round(delta))+" early"]]};
    }
    return null;
  }
