ion setSoundEnabled(on){
    soundEnabled=Boolean(on);
    const b=$("soundBtn");
    b.textContent=soundEnabled?"Sound on":"Enable sound";
    b.classList.toggle("sound-on",soundEnabled);
    b.classList.toggle("sound-off",!soundEnabled);
    updateAudioState();

    if(soundEnabled){
      await playAudioTest();
    }else{
      for(const a of activeClipAudios){try{a.pause();a.currentTime=0}catch(_){}}
      activeClipAudios=[];
    }
  }

  async function playAudioTest(){
    if(!soundEnabled)return;
    await playTags(["whistle"],.72);
    setTimeout(()=>playTags(["crowd","boo"],.70),1200);
    setTimeout(()=>playTags(["crowd","pick"],.42),3000);
  }

  function playTinyConfirm(){}

  function playPickStinger(){
    if(!soundEnabled)return;
    if(Math.random()<.56)playTags(["crowd","pick"],.38);
  }

  function playClockTick(second){
    if(!soundEnabled||second<1||second>10)return;
    const rate=.95+Math.random()*.10;
    const volume=second<=3?.72:second<=5?.58:.40;
    playTags(["tick","clock"],volume,rate);
  }

  function playPressureCue(stage){
    if(!soundEnabled)return;
    if(stage===1){
      playTags(["whistle","pressure"],.56);
    }else if(stage===2){
      if(Math.random()<.58)playTags(["crowd","gasp"],.56);
      else playTags(["crowd","aww"],.56);
    }
  }

  function playTimeoutReaction(){
    if(!soundEnabled||zeroReactionPlayed)return;
    zeroReactionPlayed=true;
    const r=Math.random();
    if(r<.50)playTags(["crowd","boo"],.82);
    else if(r<.80)playTags(["crowd","protest"],.72);
    else playTags(["crowd","aww"],.72);
  }

  function resetPressureSound(){
    pressureSoundStage=0;
    lastPressureSecond=null;
    zeroReactionPlayed=false;
  }

  function hidePressure(){
    if(!pressureShowing){
      resetPressureSound();
      return;
    }
    pressureShowing=false;
    $("pressure").className="";
    resetPressureSound();
  }

  const pressurePhraseBags={};
  let pressurePhraseTurnKey="";

  function shuffledCopy(arr){
    const x=[...arr];
    for(let i=x.length-1;i>0;i--){
      const j=Math.floor(Math.random()*(i+1));
      [x[i],x[j]]=[x[j],x[i]];
    }
    return x;
  }

  function pressurePhrase(stage,turnKey){
    if(pressurePhraseTurnKey!==turnKey)pressurePhraseTurnKey=turnKey;
    if(!pressurePhraseBags[stage]||!pressurePhraseBags[stage].length){
      pressurePhraseBags[stage]=shuffledCopy(PRESSURE_PHRASES[stage]||PRESSURE_PHRASES.ten);
    }
    return pressurePhraseBags[stage].shift();
  }

      return;

    if(!curCell){hidePressure();return;}

    const name=broadcastName(mgr);

        lastPressureSecond=whole;

      }

        pressureSoundStage=1;
        playPressureCue(1);
      }else if(seconds<=5 && seconds>0 && pressureSoundStage<2){

        playPressureCue(2);
      }else if(seconds<=0){
        playTimeoutReaction();
      }
    }
