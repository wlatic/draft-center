  function currentRoundNumber(){
    const p=liveProgress();
    return p.nextCell?.round || Number(draft?.settings?.rounds||1);
  }

  function picksForRound(round){
    return picks.filter(p=>Number(p?.round||0)===Number(round));
  }

  function positionBreakdown(arr){
    const c={QB:0,RB:0,WR:0,TE:0,K:0,DEF:0};
    for(const p of arr){
      const pos=String(p?.metadata?.position||"").toUpperCase();
      if(c[pos]!==undefined)c[pos]++;
    }
    return c;
  }

  function marketDelta(p){
    const mv=marketValue(p);
    if(!mv)return null;
    const cell=cellForPick(p);
    const ordinal=Number(cell?.ordinal||p?.pick_no||0);
    return {
      p,
      ordinal,
      adp:Number(mv.rank),
      delta:ordinal-Number(mv.rank)
    };
  }

  function currentRosterFacts(){
    const facts=[];
    const ids=Object.keys(draft?.draft_order||{});
    for(const uid of ids){
      const slot=Number(draft.draft_order[uid]);
      const mgr=managerForSlot(slot);
      const c=rosterCounts(uid);
      const total=c.QB+c.RB+c.WR+c.TE+c.K+c.DEF;
      if(!total)continue;

      if(c.WR>=4) facts.push({score:c.WR,title:`${managerTitle(mgr)}: ${c.WR} WRs`,detail:`Roster through ${total} selections`,tone:"hot"});
      if(c.RB>=4) facts.push({score:c.RB,title:`${managerTitle(mgr)}: ${c.RB} RBs`,detail:`Roster through ${total} selections`,tone:"hot"});
      if(c.QB>=2) facts.push({score:c.QB+0.5,title:`${managerTitle(mgr)}: ${c.QB} QBs`,detail:`Quarterback-heavy build`,tone:"alert"});
      if(c.TE>=2) facts.push({score:c.TE+0.25,title:`${managerTitle(mgr)}: ${c.TE} TEs`,detail:`Multiple tight ends already`,tone:"alert"});
      if(total>=5 && c.RB<=1) facts.push({score:4.4,title:`${managerTitle(mgr)}: ${c.RB} RB`,detail:`Zero-RB / light-RB build`,tone:"good"});
      if(total>=5 && c.WR<=1) facts.push({score:4.2,title:`${managerTitle(mgr)}: ${c.WR} WR`,detail:`Very light at wide receiver`,tone:"good"});
    }
    return facts.sort((a,b)=>b.score-a.score).slice(0,4);
  }

  function stackFacts(){
    const facts=[];
    const byOwner=new Map();

    for(const p of picks){
      if(isKeeperPick(p))continue;
      const uid=String(p.picked_by||"");
      if(!uid)continue;
      if(!byOwner.has(uid))byOwner.set(uid,[]);
      byOwner.get(uid).push(p);
    }

    for(const [uid,arr] of byOwner){
      const slot=Number(draft?.draft_order?.[uid]||0);
      const mgr=managerForSlot(slot);
      const byNFL={};

      for(const p of arr){
        const team=String(p?.metadata?.team||"");
        if(!team)continue;
        if(!byNFL[team])byNFL[team]=[];
        byNFL[team].push(p);
      }

      for(const [team,players] of Object.entries(byNFL)){
        const qb=players.find(p=>String(p?.metadata?.position||"").toUpperCase()==="QB");
        const catcher=players.find(p=>["WR","TE"].includes(String(p?.metadata?.position||"").toUpperCase()));
        if(qb&&catcher){
          facts.push({
            score:5,
            title:`${managerTitle(mgr)}: ${team} stack`,
            detail:`${pickName(qb)} + ${pickName(catcher)}`,
            tone:"good"
          });
        }else if(players.length>=3){
          facts.push({
            score:players.length,
            title:`${managerTitle(mgr)}: ${players.length} ${team} players`,
            detail:`Leaning heavily into one NFL team`,
            tone:"hot"
          });
        }
      }
    }

    return facts.sort((a,b)=>b.score-a.score).slice(0,4);
  }

  function positionWatchFacts(){
    const live=livePicksOnly();
    const facts=[];
    const windows=[5,8,12];

    for(const size of windows){
      const arr=live.slice(-size);
      if(arr.length<Math.min(size,4))continue;
      const c=positionBreakdown(arr);
      for(const [pos,count] of Object.entries(c)){
        if(count===0)continue;
        const threshold=["WR","RB"].includes(pos)
          ? (size===8?5:size===12?7:4)
          : (size===5?3:size===8?4:5);
        if(count>=threshold){
          facts.push({
            score:count/size,
            title:`${count} ${pos}s in last ${arr.length}`,
            detail:`${Math.round(count/arr.length*100)}% of recent picks`,
            tone:"hot"
          });
        }
      }
    }

    // Position scarcity this round
    const round=currentRoundNumber();
    const rp=picksForRound(round);
    const c=positionBreakdown(rp);
    for(const pos of ["QB","TE","DEF"]){
      if(c[pos]===0 && rp.length>=4){
        facts.push({
          score:.35,
          title:`No ${pos} yet in Round ${round}`,
          detail:`${rp.length} picks into the round`,
          tone:"good"
        });
      }
    }

    const seen=new Set();
    return facts
      .sort((a,b)=>b.score-a.score)
      .filter(f=>{const k=f.title;if(seen.has(k))return false;seen.add(k);return true;})
      .slice(0,4);
  }
